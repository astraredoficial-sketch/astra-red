========== src\lib\storage.ts ==========
import type { UserProfile } from "@/lib/types";

const KEY = "astrared:v1";

export type Store = {
  activeId: string | null;
  profiles: UserProfile[];
  passed: string[];
  liked: string[];
};

const empty = (): Store => ({
  activeId: null,
  profiles: [],
  passed: [],
  liked: [],
});

export function loadStore(): Store {
  if (typeof window === "undefined") return empty();
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return empty();
    return { ...empty(), ...JSON.parse(raw) } as Store;
  } catch {
    return empty();
  }
}

export function saveStore(store: Store): void {
  localStorage.setItem(KEY, JSON.stringify(store));
}

export function upsertProfile(profile: UserProfile): Store {
  const store = loadStore();
  const idx = store.profiles.findIndex((p) => p.id === profile.id);
  if (idx >= 0) store.profiles[idx] = profile;
  else store.profiles.push(profile);
  store.activeId = profile.id;
  saveStore(store);
  return store;
}

export function setActive(id: string): Store {
  const store = loadStore();
  store.activeId = id;
  saveStore(store);
  return store;
}

export function markSwipe(id: string, liked: boolean): Store {
  const store = loadStore();
  store.passed = store.passed.filter((x) => x !== id);
  store.liked = store.liked.filter((x) => x !== id);
  if (liked) store.liked.push(id);
  else store.passed.push(id);
  saveStore(store);
  return store;
}

export function resetSwipes(): Store {
  const store = loadStore();
  store.passed = [];
  store.liked = [];
  saveStore(store);
  return store;
}

import { useEffect, useState } from "react";
import { loadStore, type Store } from "@/lib/storage";

export function useSession() {
  const [store, setStore] = useState<Store | null>(null);

  useEffect(() => {
    setStore(loadStore());
  }, []);

  const me = store?.profiles.find((p) => p.id === store.activeId) ?? null;
  return { ready: store !== null, store, me, refresh: () => setStore(loadStore()) };
}

========== src\lib\astro\time.ts ==========
export function civilToUtc(
  date: string,
  time: string,
  timeZone: string,
): Date {
  const [year, month, day] = date.split("-").map(Number);
  const [hour, minute] = time.split(":").map(Number);
  const desired = Date.UTC(year, month - 1, day, hour, minute, 0);

  const offsetAt = (instant: number) => {
    const parts = Object.fromEntries(
      new Intl.DateTimeFormat("en-US", {
        timeZone,
        year: "numeric",
        month: "2-digit",
        day: "2-digit",
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
        hourCycle: "h23",
      })
        .formatToParts(new Date(instant))
        .filter((p) => p.type !== "literal")
        .map((p) => [p.type, p.value]),
    );
    const asUtc = Date.UTC(
      Number(parts.year),
      Number(parts.month) - 1,
      Number(parts.day),
      Number(parts.hour),
      Number(parts.minute),
      Number(parts.second),
    );
    return asUtc - instant;
  };

  const first = offsetAt(desired);
  let instant = desired - first;
  const second = offsetAt(instant);
  if (second !== first) instant = desired - second;
  return new Date(instant);
}

import * as Astronomy from "astronomy-engine";
import { findCity } from "@/lib/cities";
import type { BirthData, NatalChart, Placement, PlanetId } from "@/lib/types";
import { signFromLongitude } from "./constants";
import { civilToUtc } from "./time";

const BODIES: { id: Exclude<PlanetId, "asc" | "node">; body: Astronomy.Body }[] =
  [
    { id: "sun", body: Astronomy.Body.Sun },
    { id: "moon", body: Astronomy.Body.Moon },
    { id: "mercury", body: Astronomy.Body.Mercury },
    { id: "venus", body: Astronomy.Body.Venus },
    { id: "mars", body: Astronomy.Body.Mars },
    { id: "jupiter", body: Astronomy.Body.Jupiter },
    { id: "saturn", body: Astronomy.Body.Saturn },
    { id: "uranus", body: Astronomy.Body.Uranus },
    { id: "neptune", body: Astronomy.Body.Neptune },
    { id: "pluto", body: Astronomy.Body.Pluto },
  ];

function norm(deg: number): number {
  return ((deg % 360) + 360) % 360;
}

function eclipticLongitude(body: Astronomy.Body, time: Astronomy.AstroTime): number {
  if (body === Astronomy.Body.Moon) {
    return norm(Astronomy.EclipticGeoMoon(time).lon);
  }
  const vec = Astronomy.GeoVector(body, time, true);
  return norm(Astronomy.Ecliptic(vec).elon);
}

function meanNorthNode(time: Astronomy.AstroTime): number {
  const T = time.tt / 36525;
  return norm(125.04452 - 1934.136261 * T + 0.0020708 * T * T + T ** 3 / 450000);
}

function ramcToMc(ramcDeg: number, obliquityDeg: number): number {
  const ramc = (ramcDeg * Math.PI) / 180;
  const eps = (obliquityDeg * Math.PI) / 180;
  return norm(
    (Math.atan2(Math.sin(ramc), Math.cos(ramc) * Math.cos(eps)) * 180) / Math.PI,
  );
}

function ascendant(ramcDeg: number, latDeg: number, obliquityDeg: number): number {
  const ramc = (ramcDeg * Math.PI) / 180;
  const lat = (latDeg * Math.PI) / 180;
  const eps = (obliquityDeg * Math.PI) / 180;
  const y = Math.cos(ramc);
  const x = -(Math.sin(ramc) * Math.cos(eps) + Math.tan(lat) * Math.sin(eps));
  return norm((Math.atan2(y, x) * 180) / Math.PI);
}

function houseOf(longitude: number, cusps: number[]): number {
  for (let i = 0; i < 12; i++) {
    const a = cusps[i];
    const b = cusps[(i + 1) % 12];
    if (a < b) {
      if (longitude >= a && longitude < b) return i + 1;
    } else if (longitude >= a || longitude < b) {
      return i + 1;
    }
  }
  return 1;
}

function placement(id: PlanetId, longitude: number, houses: number[]): Placement {
  const sign = signFromLongitude(longitude);
  return {
    id,
    longitude,
    sign: sign.sign,
    signIndex: sign.signIndex,
    degreeInSign: sign.degreeInSign,
    house: houseOf(longitude, houses),
  };
}

export function computeNatalChart(birth: BirthData): NatalChart {
  const city = findCity(birth.cityId);
  if (!city) throw new Error("Ciudad no encontrada");

  const timeStr = birth.timeUnknown ? "12:00" : birth.time;
  const utc = civilToUtc(birth.date, timeStr, city.tz);
  const astroTime = Astronomy.MakeTime(utc);
  const obliquity = 23.4392911 - (46.815 / 3600) * (astroTime.tt / 36525);

  const gastHours = Astronomy.SiderealTime(astroTime);
  const lstHours = ((gastHours + city.lng / 15) % 24 + 24) % 24;
  const ramc = lstHours * 15;
  const ascLon = ascendant(ramc, city.lat, obliquity);
  const mcLon = ramcToMc(ramc, obliquity);
  void mcLon;

  const houses = Array.from({ length: 12 }, (_, i) => norm(ascLon + i * 30));

  const planets: Placement[] = BODIES.map(({ id, body }) =>
    placement(id, eclipticLongitude(body, astroTime), houses),
  );
  const node = placement("node", meanNorthNode(astroTime), houses);
  const asc = placement("asc", ascLon, houses);
  planets.push(node, asc);

  const sun = planets.find((p) => p.id === "sun")!;
  const moon = planets.find((p) => p.id === "moon")!;

  return {
    generatedAt: new Date().toISOString(),
    julianDay: astroTime.ut + 2451545.0,
    houses,
    planets,
    triad: { sun, moon, asc },
    timeUnknown: birth.timeUnknown,
  };
}

import type {
  AspectName,
  NatalChart,
  PlanetId,
  SynastryAspect,
  SynastryResult,
} from "@/lib/types";
import { angleDelta, ASPECT_META } from "./constants";

const PERSONAL: PlanetId[] = ["sun", "moon", "venus", "mars", "asc", "mercury"];

function pairWeight(a: PlanetId, b: PlanetId): number {
  const key = [a, b].sort().join("-");
  const weights: Record<string, number> = {
    "moon-sun": 14,
    "mars-venus": 12,
    "asc-asc": 10,
    "sun-sun": 9,
    "moon-moon": 9,
    "venus-venus": 8,
    "sun-venus": 7,
    "asc-sun": 7,
    "asc-moon": 7,
    "moon-venus": 7,
    "mars-sun": 6,
    "node-sun": 9,
    "moon-node": 9,
    "node-venus": 8,
    "asc-node": 8,
    "mars-node": 6,
  };
  return weights[key] ?? 3;
}

function aspectHarmony(type: AspectName): number {
  switch (type) {
    case "trine":
    case "sextile":
      return 1;
    case "conjunction":
      return 0.92;
    case "opposition":
      return 0.58;
    case "square":
      return 0.48;
  }
}

function findAspect(delta: number): { type: AspectName; orb: number } | null {
  for (const type of Object.keys(ASPECT_META) as AspectName[]) {
    const meta = ASPECT_META[type];
    const orb = Math.abs(delta - meta.angle);
    if (orb <= meta.orb) return { type, orb };
  }
  return null;
}

export function computeSynastry(a: NatalChart, b: NatalChart): SynastryResult {
  const aspects: SynastryAspect[] = [];
  let scored = 0;
  let max = 0;

  for (const pa of a.planets) {
    for (const pb of b.planets) {
      if (pa.id === "asc" && pb.id !== "asc" && !PERSONAL.includes(pb.id)) continue;
      const weight = pairWeight(pa.id, pb.id);
      max += weight;
      const delta = angleDelta(pa.longitude, pb.longitude);
      const hit = findAspect(delta);
      if (!hit) continue;
      const tightness = 1 - hit.orb / ASPECT_META[hit.type].orb;
      scored += weight * tightness * aspectHarmony(hit.type);
      aspects.push({
        a: pa.id,
        b: pb.id,
        type: hit.type,
        orb: Number(hit.orb.toFixed(2)),
        exactAngle: Number(delta.toFixed(2)),
      });
    }
  }

  const ratio = max > 0 ? scored / max : 0;
  const percent = Math.round(Math.min(99, Math.max(18, 28 + ratio * 95)));

  const hasNodalAspect = aspects.some((asp) => {
    const nodal = asp.a === "node" || asp.b === "node";
    const personal =
      PERSONAL.includes(asp.a) || PERSONAL.includes(asp.b);
    const contact = asp.type === "conjunction" || asp.type === "opposition";
    return nodal && personal && contact && asp.orb <= 6;
  });

  aspects.sort((x, y) => pairWeight(y.a, y.b) - pairWeight(x.a, x.b));

  return {
    percent,
    aspects: aspects.slice(0, 12),
    hasNodalAspect,
    hiloRojo: percent > 90 && hasNodalAspect,
  };
}
